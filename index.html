  <!-- Firebase Logic -->

  <script type="module">

    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";

    import { getAuth, signInWithEmailAndPassword, createUserWithEmailAndPassword, signOut, onAuthStateChanged } 

      from "https://www.gstatic.com/firebasejs/10.12.0/firebase-auth.js";

    import { getFirestore, doc, getDoc, setDoc } 

      from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js";


    // 1. Replace with your actual Firebase project config

    const firebaseConfig = {

      apiKey: "YOUR_API_KEY",

      authDomain: "YOUR_PROJECT_ID.firebaseapp.com",

      projectId: "YOUR_PROJECT_ID",

      storageBucket: "YOUR_PROJECT_ID.appspot.com",

      messagingSenderId: "YOUR_SENDER_ID",

      appId: "YOUR_APP_ID"

    };


    const app = initializeApp(firebaseConfig);

    const auth = getAuth(app);

    const db = getFirestore(app);


    // 2. Sample Concept Nodes (Multiple questions per node)

    const CONCEPT_NODES = [

      {

        id: "concept_vf_post_shock",

        category: "ACLS - VF/pVT",

        answer: "חזרה מיידית ל-2 דקות עיסויי חזה (CPR) והשגת נתיב ורידי/תוך-גרמי (IV/IO).",

        prompts: [

          { type: "תרחיש קליני", text: "נתת שוק ראשון למטופל עם VF. המסך התאפס. מה הפעולה המיידית הבאה?" },

          { type: "שלב באלגוריתם", text: "סיימת מתן שוק 1 ב-pulseless VT. איזו פעולה פולשנית יש לבצע במהלך סבב ה-CPR הנוכחי?" },

          { type: "נכון / לא נכון", text: "נכון/לא נכון: מיד לאחר שוק 1 ב-VF, יש לעצור עיסויים ל-5 שניות לבדיקת דופק ומקצב." }

        ]

      },

      {

        id: "concept_epi_timing",

        category: "ACLS - תרופות",

        answer: "מתן אפינפרין (1 מ\"ג כל 3-5 דקות) החל מסבב ה-CPR השני (לאחר שוק שני).",

        prompts: [

          { type: "תזמון תרופתי", text: "המטופל נותר ב-VF לאחר שוק 1 ו-2 דקות CPR. ניתן שוק 2. איזו תרופה ניתנת כעת במהלך סבב ה-CPR?" },

          { type: "זיהוי הפוך", text: "במהלך החייאת VF shockable, ניתנה מנת האפינפרין הראשונה. כמה שוקים ניתנו במצטבר עד לרגע זה?" }

        ]

      }

    ];


    // State Variables

    let currentUser = null;

    let userProgress = {}; // { [conceptId]: { repetitions: 0, nextReview: Date } }

    let currentConcept = null;


    // UI Elements

    const authSection = document.getElementById("auth-section");

    const studySection = document.getElementById("study-section");

    const userNav = document.getElementById("user-nav");

    const userDisplay = document.getElementById("user-display");

    const authError = document.getElementById("auth-error");

    const cardContainer = document.getElementById("flashcard-container");

    const promptText = document.getElementById("prompt-text");

    const promptType = document.getElementById("prompt-type");

    const answerSection = document.getElementById("answer-section");

    const answerText = document.getElementById("answer-text");

    const btnReveal = document.getElementById("btn-reveal");


    // 3. Auth Handlers

    document.getElementById("login-btn").onclick = async () => {

      authError.classList.add("hidden");

      try {

        await signInWithEmailAndPassword(auth, document.getElementById("auth-email").value, document.getElementById("auth-password").value);

      } catch (err) {

        authError.textContent = err.message;

        authError.classList.remove("hidden");

      }

    };


    document.getElementById("signup-btn").onclick = async () => {

      authError.classList.add("hidden");

      try {

        await createUserWithEmailAndPassword(auth, document.getElementById("auth-email").value, document.getElementById("auth-password").value);

      } catch (err) {

        authError.textContent = err.message;

        authError.classList.remove("hidden");

      }

    };


    document.getElementById("logout-btn").onclick = () => signOut(auth);


    onAuthStateChanged(auth, async (user) => {

      if (user) {

        currentUser = user;

        userDisplay.textContent = user.email;

        authSection.classList.add("hidden");

        studySection.classList.remove("hidden");

        userNav.classList.remove("hidden");

        await loadUserData();

        loadNextCard();

      } else {

        currentUser = null;

        authSection.classList.remove("hidden");

        studySection.classList.add("hidden");

        userNav.classList.add("hidden");

      }

    });


    // 4. Firestore Sync

    async function loadUserData() {

      const userRef = doc(db, "users", currentUser.uid);

      const docSnap = await getDoc(userRef);

      if (docSnap.exists()) {

        userProgress = docSnap.data().progress || {};

      } else {

        userProgress = {};

        await setDoc(userRef, { progress: {} });

      }

    }


    async function saveProgression(conceptId, wasCorrect) {

      if (!userProgress[conceptId]) {

        userProgress[conceptId] = { reps: 0 };

      }

      userProgress[conceptId].reps = wasCorrect ? userProgress[conceptId].reps + 1 : 0;

      await setDoc(doc(db, "users", currentUser.uid), { progress: userProgress }, { merge: true });

    }


    // 5. Concept & Layout Randomization

    function applyVisualPerturbation() {

      const borders = ['border-blue-300', 'border-emerald-300', 'border-slate-300', 'border-purple-300'];

      cardContainer.className = "bg-white p-6 rounded-2xl shadow-sm transition-all duration-200 border-2 " + 

        borders[Math.floor(Math.random() * borders.length)];

    }


    function loadNextCard() {

      answerSection.classList.add("hidden");

      btnReveal.classList.remove("hidden");


      // Pick random concept

      currentConcept = CONCEPT_NODES[Math.floor(Math.random() * CONCEPT_NODES.length)];

      // Pick random prompt within that concept

      const prompt = currentConcept.prompts[Math.floor(Math.random() * currentConcept.prompts.length)];


      promptType.textContent = `${currentConcept.category} • ${prompt.type}`;

      promptText.textContent = prompt.text;

      answerText.textContent = currentConcept.answer;


      const reps = userProgress[currentConcept.id]?.reps || 0;

      document.getElementById("concept-rep").textContent = reps;


      applyVisualPerturbation();

    }


    btnReveal.onclick = () => {

      btnReveal.classList.add("hidden");

      answerSection.classList.remove("hidden");

    };


    document.getElementById("btn-pass").onclick = async () => {

      await saveProgression(currentConcept.id, true);

      loadNextCard();

    };


    document.getElementById("btn-fail").onclick = async () => {

      await saveProgression(currentConcept.id, false);

      loadNextCard();

    };

  </script>
